// --- UPDATED DASH & INPUT CONTROLS --- // Add inside GameScene create() to prevent Space/Shift from scrolling the page this.input.keyboard.addCapture([ Phaser.Input.Keyboard.KeyCodes.SPACE, Phaser.Input.Keyboard.KeyCodes.SHIFT, Phaser.Input.Keyboard.KeyCodes.W, Phaser.Input.Keyboard.KeyCodes.A, Phaser.Input.Keyboard.KeyCodes.S, Phaser.Input.Keyboard.KeyCodes.D ]); // Updated update() method update() { let vx = 0; let vy = 0; // Read inputs if (this.wasd.up.isDown || this.cursors.up.isDown) vy -= 1; if (this.wasd.down.isDown || this.cursors.down.isDown) vy += 1; if (this.wasd.left.isDown || this.cursors.left.isDown) vx -= 1; if (this.wasd.right.isDown || this.cursors.right.isDown) vx += 1; const moveVec = new Phaser.Math.Vector2(vx, vy).normalize(); // Check for Dash Input BEFORE updating regular velocity const dashPressed = Phaser.Input.Keyboard.JustDown(this.wasd.space) || Phaser.Input.Keyboard.JustDown(this.wasd.shift); if (dashPressed && (vx !== 0 || vy !== 0)) { this.localPlayer.dash(moveVec); } // ONLY apply regular movement velocity if the player is NOT currently dashing if (!this.localPlayer.isDashing) { this.localPlayer.body.setVelocity( moveVec.x * this.localPlayer.stats.speed, moveVec.y * this.localPlayer.stats.speed ); } // Run distance tag check every frame this.checkTagLogic(); this.localPlayer.renderAura(); if (this.isVSBot) { this.remotePlayer.updateAI(this.localPlayer); } }
